-- bash shell expansions - by order of precedence
	1. Brace Expansion 	# note: it DOES NOT have "$" prefix as with parameter expansion
		echo {1..3}
		touch trade{1,2,3}.csv          # creates ALL 3 files
		echo t{1,2,3,4}					# t1 t2 t3 t4
		echo t{01..12}					# t01 t02 t03 t04 t05 t06 t07 t08 t09 t10 t11 t12
		echo {10..0..-2}         		# 10 8 6 4 2 0  -> 0 to 10 decrement by 2
		echo equity/{daily,hourly}.csv  # equity/daily.csv equity/hourly.csv
		echo {a..f}          			# a b c d e f


	2. Tilde Expansion ~
	3. Parameter Expansion # ${var}, ${var#pattern}, $var  
			# Parameter Expansion ${var#pattern} - String Manipulation (Later in expansion order)		
				Syntax			Effect							Example
				${var#prefix}	Remove shortest prefix match	${logfile#*/}
				${var##prefix}	Remove longest prefix match		${logfile##*/}
				${var%suffix}	Remove shortest suffix match	${logfile%.csv}
				${var%%suffix}	Remove longest suffix match		${logfile%%.*}
				${var:0:3}" # Outputs: first three chats

				examples
					logfile="/var/log/trades/equity_20260214.csv"
					echo ${logfile##*/}      # equity_20260214.csv
					echo ${logfile%.*}       # /var/log/trades/equity_20260214
					echo ${logfile##equity_} # 20260214.csv
					echo ${logfile#*/}       # log/trades/equity_20260214.csv			

	4. Command Substitution $(cmd) `cmd`
	5. Arithmetic Expansion $((2+2))
	6. Word Splitting (IFS)
	7. Filename Expansion * ? []  # globbing
		ls trade*.csv     # trade1.csv trade2.csv
		ls trade?.csv     # trade1.csv (single char)
		ls trade[1-3].csv # trade1.csv trade2.csv trade3.csv 	# GLOB: list only existing files (unlike brace expansion)
	8. Quote Removal
